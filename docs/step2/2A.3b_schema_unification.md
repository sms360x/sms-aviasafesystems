## 7. Rollback procedure

```sql

DELETE FROM public.tenants
 WHERE kind = 'regulator'
   AND id IN ('56cf5288-56a5-5188-a56d-e3442492fedd',
              '53708c97-e4cf-5605-a1eb-a131889a8cc1');
-- 2. Drop the added columns.

  DROP COLUMN IF EXISTS operator_tenant_ids,
  DROP COLUMN IF EXISTS short_name,

  DROP COLUMN IF EXISTS country_name,
  DROP COLUMN IF EXISTS domain,
  DROP COLUMN IF EXISTS is_saas_customer,
  DROP COLUMN IF EXISTS subscription_status,

  DROP COLUMN IF EXISTS subscription_end,
  DROP COLUMN IF EXISTS regulator_type,
  DROP COLUMN IF EXISTS display_name,
  DROP COLUMN IF EXISTS website;
-- 3. Drop the kind index + column.

ALTER TABLE public.tenants DROP COLUMN IF EXISTS kind;
-- 4. Reverse the contact_* type alignment (2A.3b-fix).
ALTER TABLE public.tenants
  ALTER COLUMN contact_email TYPE text,
  ALTER COLUMN contact_phone TYPE text;
```
